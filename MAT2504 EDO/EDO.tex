\subsection{Generalidades y Definiciones}
\fecha{Clase 1. Jueves 6/8.}

Las leyes y los fenómenos del universo están escritos en el lenguaje matemático, y uno de los principales problemas de los científicos antiguos fué el estudio de los movimientos planetarios.

\begin{definition}
    Una \wea{ecuación diferencial ordinaria} (\wea{EDO}) de \wea{orden $k$} es una relación funcional entre la variable real $t \in \R$, una función $t \mapsto y(t)$ y sus derivadas $y', y'', \dots, y^{(k)}$ en $t$, definida por
    \[ F(t,y,y',y'',\dots,y^{(k)}) = 0, \tag{1} \]
    donde $y \colon J \to \R^m$ y $J \subset \R$ es un intervalo abierto y 
    \[ F \colon J \times \R^m \times \dots \times \R^m \to \R^n \]
    es una función dada ($F \colon J \times \R^{(k+1)m} \to \R^n$).
    Generalmente $t$ se llama la \wea{variable independiente} e $y$ la \wea{variable dependiente}.
\end{definition}

\begin{example}\
    \begin{itemize}
        \item $f' + 2f = 10$ es una EDO, con la variable independiente dada implícitamente. La relación funcional es
        \[ F(x,f,f') \mapsto f' + 2f - 10. \]
        \item $f' + 2f = e^x \sin x$ es una EDO, con relación funcional
        \[ F(x,f,f') \mapsto f' + 2f - e^x \sin x. \]
    \end{itemize}
\end{example}

\begin{definition}
    Si $F$ es una función a valores reales, la EDO se dice \wea{ecuación escalar}.
\end{definition}

\begin{obs}
    En general $m = 1$, es decir $y \colon J \to \R$.
\end{obs}

\begin{example}
    \
    \begin{itemize}
        \item $ty' + 5y^2 = 5t^3$.
        \item $t^2 y'' - 2(y')^2 = 0$.
    \end{itemize}
\end{example}

\begin{notation}
$C^k(U,V)$ es el conjunto de las funciones $f \colon U \to V$ de clase $C^k$. Usamos también las notaciones
\[ y^{(n)}(t) = \dv{^n y}{t^n}. \]
\end{notation}

\begin{definition}
    Si $F$ no es independiente en $y^{(k)}$, entonces diremos que
    \[ y^{(k)} = f(t,y,y',\dots,y^{(k-1)}) \]
    es un sistema de $m \times m$ ecuaciones diferenciales ordinarias de orden $k$ con respecto a la función desconocida $y \colon J \to \R^m$ y  donde la función $f \colon J \times \R^{km} \to \R^m$. Escribimos
    \[ y_i^{(k)} = f_i(t,y,y',y^{(k-1)}), \quad \forall i \in \{1,\dots,m\} \]
    donde $y = (y_1,\dots,y_m)$ y $f = (f_1,\dots,f_m)$.
\end{definition}

\begin{example}
    \[ ty + (1+t^2) y' + \sin t y'' = 0. \]
    En este caso tenemos $F \colon (t,y,y',y'') \mapsto ty + (1+t^2) y' + \sin t y''$.
    Si consideramos momentáneamente valores de $t$ tales que $\sin t \neq 0$, entonces podemos reescribir esta EDO de la forma
    \[ y'' = - \frac{t}{\sin t} y - \frac{(1+t^2)}{\sin t} y'. \]
    Si tomamos $\ds f \colon (x,y,y') \mapsto - \frac{t}{\sin t} y - \frac{(1+t^2)}{\sin t} y'$, entonces pasamos el sistema de la forma $F(t,y,y',y'') = 0$ a $y'' = f(t,y,y')$ (forma estándar).
\end{example}

\begin{obs}
    \ \begin{itemize}
        \item En una EDO, la función desconocida $y$ depende de una sola variable independiente.
        \item Si la variable dependiente $y$ es una función de dos o más variables independientes, entonces aparecerán \wea{derivadas parciales} y hablaremos de EDP, ecuaciones a derivadas parciales. Por ejemplo, la temperatura $u(t,x)$ de una barra uniforme en el punto $x$ al tiempo $t$ satisface (junto con algunas condiciones adicionales) la siguiente EDP:
        \[ \frac{\del u}{\del t} = k \frac{\del^2 u}{\del x^2}, \]
        donde $k$ es la difusividad térmica de la barra (cst).
    \end{itemize}
\end{obs}

\begin{notation}
    Para derivadas parciales, escribimos
    \[ \frac{\del f}{\del x} = \del_x f. \]
\end{notation}

\begin{definition}
    Si $f \colon U \subset \R^n \to \R^m$, escribimos
    \[ \d f = \del_{x_1} f \d x_1 + \dots \del_{x_n} f \d x_n. \]
\end{definition}

\subsection{Solución de una ecuación diferencial}

Consideremos la EDO
\[ 2 f'' + f + (x^2 + 1) f = 0. \]
¿Cómo podemos resolverla? ¿Qué significa resolver una EDO? Queremos encontrar una función $f$ que satisfaga la relación funcional. También queremos saber en donde está definida, y claramente que sea de clase $C^2$ en su dominio.

\begin{definition}
    Una función $\phi \in C^k(I,\R^m)$ es \wea{solución} de la EDO (1) sobre $I \subseteq J$ si tenemos
    \[ \forall t \in I, \quad F(t,\phi(t),\phi'(t), \dots, \phi^{(k)}(t)) = 0. \]
    En este caso hablamos de una solución \wea{explícita}.
\end{definition}

\begin{example}
    La función $y(x) = 7 e^x + x^2 + 2x + 2$ es una solución de la EDO $y' = y - x^2$ en $I = \R$, ya que efectivamente
    \[ y'(x) = 7e^x + 2x + 2 = y(x) - 2. \]
\end{example}

\begin{definition}
    Una EDO de orden $k$ se dice de \wea{forma canónica} (\wea{normal}) si se escribe de la forma
    \[ y(k) = f(t,y,y',\dots,y^{(k-1)}), \tag{2} \]
    donde $f \colon I \times \R^{km} \to \R^m$ e $I \subset J$.
\end{definition}

\fecha{Clase 2. Martes 11/8.}

\begin{obs}
    No siempre se puede escribir una EDO de forma canónica.
    \begin{itemize}
        \item Puede ser localmente posible a través del Teorema de la Función Implícita.
        \item $f$ puede ser que no sea definida como función. Por ejemplo considerando $(y')^2 - t = 0$.
        \item La forma canónica es una propiedad que una EDO puede no tener.
    \end{itemize}
\end{obs}

\begin{definition}
    La ecuación (1) se llama \wea{lineal} si $$f \colon (t,x_1,x_2,\dots,x_n) \to f(t,x_1,x_2,\dots,x_n)$$ es una función afín con respecto a la variable $x = (x_1,x_2,\dots,x_n)$. Es decir,
    \[ y^{(k)}(t) = \sum_{j=0}^{k-1} a_j(t) y^{(j)}(t) + g(t) \]
    donde $a_j$ son funciones matrices $m \times m$ definidas en $J$ y $g \colon J \to \R^m$.
    Si $g \equiv 0$, el sistema se dice \wea{homogeneo}.
\end{definition}

\begin{example}
    $f(x_1,x_2) = x_1 + 2x_2 - x_1 x_2 - 100$ funciona, pero $g(x_1,x_2) = 2x_1 - \sin x_2$ no funciona.
\end{example}

\begin{obs}
    En el caso de una EDO lineal, si escribimos $y = \matr{y_1 \\ \vdots \\ y_m}$, entonces
    \begin{align*}
        y^{(k)} &= f(t,y,y',\dots,y^{(k-1)}) \\
        &= \sum_{j=0}^{k-1} a_j(t) y^{(j)}(t) + g(t) \\
        &= \matr{a_0(t) & \cdots & a_{k-1}(t)} \matr{y(t) \\ y'(t) \\ \vdots \\ y^{(k-1)}(t)} + \matr{g_1(t) \\ g_2(t) \\ \vdots \\ g_{k-1}(t)}.
    \end{align*}
\end{obs}

\begin{example}

        \item $y'' = 2yt^2$.
    \end{itemize}
\end{example}

\end{example}

\begin{exercise}
    [Invarianza]
    Si la función
    \begin{align*}
        \phi \colon I &\to U \\
        t &\mapsto \phi(t)
    \end{align*}
    es solución de la ecuación autónoma $y' = f(y)$ en un intervalo abierto $I \subseteq \R$, entonces
    \begin{align*}
        \psi_a \colon I + a &\to U \\
        t &\mapsto \phi(t-a)
    \end{align*}
    para $a \in \R$ es también solución de $y' = f(y)$.
\end{exercise}
\begin{proof}
    Sea $\psi_a(t) = \phi(t-a)$, entonces
    \[ \psi_a'(t) = \dv{}{t} \phi(t-a) = \phi'(t-a) \cdot \dv{}{t} (t-a) = \phi'(t-a). \]
    Notemos que para todo $t \in I+a$ tenemos que $t-a \in I$, y como $\phi$ es solución de la EDO en $I$, tenemos que
    \[ \psi'(t) = \phi'(t-a) = f(\phi(t-a)) = f(\psi(t)). \]
    Luego $\psi$ es solución de la EDO en el intervalo $I+a$.
\end{proof}

\end{obs}

\begin{definition}
    Sean $J \subset \R$ un intervalo abierto y $U \subset \R^m$ abierto. Sea $f \colon J \times U \to \R^m$ una aplicación continua. Una \wea{solución} de la ecuación diferencial
    \[ x' = f(t,x) \]
    es una aplicación $x \colon I \subset J \to \R^m$ de clase $C^1$ tal que
    \[ \forall t \in I, \quad x'(t) = f(t,x(t)). \]
    Una aplicación $f \colon J \times U \to \R^m$ se llama también un \wea{campo de vectores} donde
    \begin{itemize}
        \item la variable $t \in J$ corresponde al \wea{tiempo},
        \item la variable $x \in U$ corresponde a la \wea{posición}.
    \end{itemize}
\end{definition}

En cada instante $t \in J$ y en cada punto $x \in U$, tenemos un vector $f(t,x) \in \R^m$. Podemos imaginar que el vector $f(t,x) \in \R^m$ indique el viento (velocidad de dirección) al instante $t$ en el punto $x$. Resolver la ED
\[ x' = f(t,x) \]
es determinar el comportamiento $t \mapsto x(t)$ de una partícula transportada por el viento y cuyo vector velocidad en cada instante $t$ es dado por $x'(t) = f(t,x(t))$.
